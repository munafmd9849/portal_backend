# Frontend Environment Variables
# ===============================

# API Base URL
# Development default: http://localhost:3000/api
# Production: Set to your backend API URL
VITE_API_URL=http://localhost:3000/api

# Socket.IO URL
# Development default: http://localhost:3000
# Production: Set to your backend server URL
VITE_SOCKET_URL=http://localhost:3000

# Frontend URL (optional)
# Used for OAuth redirects and link generation
VITE_FRONTEND_URL=http://localhost:5173

# Note: All Vite environment variables must be prefixed with VITE_
# They are exposed to the client-side code, so don't include sensitive secrets
